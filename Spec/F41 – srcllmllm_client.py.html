class LLMProvider(str, Enum):
    OLLAMA = "ollama"
    OPENAI = "openai"

class LLMModelType(str, Enum):

@dataclass
class LLMRequest:
    id: str = field(default_factory=lambda: hashlib.md5(str(datetime.utcnow().timestamp()).encode()).hexdigest()[:8])
    prompt: str = ""
    system_prompt: Optional[str] = None
    temperature: float = 0.1
    max_tokens: Optional[int] = None
    model: Optional[str] = None
    provider: Optional[str] = None
    timestamp: datetime = field(default_factory=datetime.utcnow)
    metadata: Dict[str, Any] = field(default_factory=dict)
    
    def to_dict(self) -> Dict[str, Any]:
        return {
            "id": self.id,
            "prompt": self.prompt[:200] + "..." if len(self.prompt) > 200 else self.prompt,
            "system_prompt": self.system_prompt[:200] + "..." if self.system_prompt and len(self.system_prompt) > 200 else self.system_prompt,
            "temperature": self.temperature,
            "max_tokens": self.max_tokens,
            "model": self.model,
            "provider": self.provider,
            "timestamp": self.timestamp.isoformat(),
            "metadata": self.metadata
        }

@dataclass

@dataclass
class LLMMetrics:

    total_tokens: int = 0

class LLMClient(ABC):
    def __init__(
        self,
        model: str,
        temperature: float = 0.1,
        timeout: int = 60,
        provider: str = "unknown",
        cache_enabled: bool = False,
        cache_ttl: int = 3600
    ):
        self.model = model
        self.temperature = temperature
        self.timeout = timeout
        self.provider = provider
        self.cache_enabled = cache_enabled
        self.cache_ttl = cache_ttl
        
        self._last_request_time: Optional[datetime] = None
        self._request_count = 0
        self._metrics = LLMMetrics()
        self._callbacks: List[Callable[[LLMRequest, Optional[LLMResponse], Optional[Exception]], Awaitable[None]]] = []
        self._cache: Dict[str, Dict[str, Any]] = {}

    @abstractmethod
    async def generate(self, prompt: str, system_prompt: Optional[str] = None, **kwargs) -> str:
        pass

    @abstractmethod
    async def generate_stream(self, prompt: str, system_prompt: Optional[str] = None, **kwargs) -> AsyncIterator[str]:
        pass

    @abstractmethod
    async def embed(self, text: str) -> List[float]:
        pass

    @abstractmethod
    async def health_check(self) -> bool:
        pass

    @abstractmethod
    async def list_models(self) -> List[str]:
        pass

    def _generate_cache_key(self, prompt: str, system_prompt: Optional[str] = None, **kwargs) -> str:
        cache_data = {
            "prompt": prompt,
            "system_prompt": system_prompt,
            "temperature": kwargs.get("temperature", self.temperature),
            "model": self.model,
            "kwargs": kwargs
        }
        return hashlib.md5(json.dumps(cache_data, sort_keys=True).encode()).hexdigest()

    async def generate_with_cache(self, prompt: str, system_prompt: Optional[str] = None, **kwargs) -> str:
        cache_key = self._generate_cache_key(prompt, system_prompt, **kwargs)
        if self.cache_enabled and cache_key in self._cache:
            entry = self._cache[cache_key]
            if (datetime.utcnow() - entry["timestamp"]).total_seconds() < self.cache_ttl:
                return entry["response"]

        response_text = await self.generate(prompt=prompt, system_prompt=system_prompt, **kwargs)
        
        if self.cache_enabled:
            self._cache[cache_key] = {"response": response_text, "timestamp": datetime.utcnow()}
        return response_text

    def add_callback(self, callback: Callable) -> None:
        self._callbacks.append(callback)

    def get_metrics(self) -> LLMMetrics:
        return self._metrics

    def get_stats(self) -> Dict[str, Any]:
        return {
            "model": self.model,
            "provider": self.provider,
            "cache_size": len(self._cache),
            "metrics": self._metrics.to_dict()
        }

# Utilisation du client
client = OllamaClient(
    model="deepseek-coder:6.7b-instruct",
    temperature=0.3,
    cache_enabled=True,
    cache_ttl=3600
)

# Ajout d'un callback
async def log_request(request, response, error):
    print(f"Request: {request.id}, Success: {response is not None}")

client.add_callback(log_request)

# Génération avec cache
response = await client.generate_with_cache(
    prompt="Write a Solidity function",
    system_prompt="You are an expert",
    max_tokens=500
)

# Métriques
metrics = client.get_metrics()
print(f"Total requests: {metrics.total_requests}")
print(f"Total cost: ${metrics.total_cost:.4f}")

# Statistiques
stats = client.get_stats()
print(f"Cache size: {stats['cache_size']}")
